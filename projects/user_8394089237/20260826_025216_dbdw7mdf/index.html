import os import subprocess import threading from flask import Flask, render_template_string, request, redirect, url_for app = Flask(__name__) UPLOAD_FOLDER = 'uploads' os.makedirs(UPLOAD_FOLDER, exist_ok=True) # لتتبع العمليات والملفات المحفوظة running_processes = {} process_logs = {} HTML_TEMPLATE = """ Nexus Advanced Panel

⚡ NEXUS PRO PANEL (STABLE) ⚡

📦 تثبيت مكتبة عبر PIP

📂 رفع سكربت جديد (.py)

⚙️ السكربتات المتاحة على السيرفر

{% for file in files %} {% endfor %}
اسم الملف الحالة التحكم والإدارة
{{ file }} {% if file in running_processes and running_processes[file].poll() is none %} ● يعمل {% else %} ○ متوقف {% endif %} {% if file not in running_processes or running_processes[file].poll() is not none %} تشغيل {% else %} إيقاف {% endif %} حذف

🖥️ سجل الأخطاء والمخرجات (Console Log)

{{ console_log }}
""" last_log = "جاهز... يرجى رفع السكربت وتثبيت المكاتب اللازمة له أولاً." @app.route('/') def index(): files = [f for f in os.listdir(UPLOAD_FOLDER) if f.endswith('.py')] return render_template_string(HTML_TEMPLATE, files=files, running_processes=running_processes, console_log=last_log) @app.route('/upload', methods=['POST']) def upload_file(): global last_log if 'file' in request.files: file = request.files['file'] if file.filename != '': filepath = os.path.join(UPLOAD_FOLDER, file.filename) file.save(filepath) last_log = f"تم رفع الملف '{file.filename}' بنجاح وجاهز للتشغيل." return redirect(url_for('index')) @app.route('/install', methods=['POST']) def install_package(): global last_log pkg = request.form.get('package_name') try: process = subprocess.run(['pip', 'install', pkg], capture_output=True, text=True, timeout=120) if process.returncode == 0: last_log = f"✅ تم تثبيت المكتبة '{pkg}' بنجاح!\n\n{process.stdout}" else: last_log = f"❌ فشل تثبيت المكتبة:\n{process.stderr}" except Exception as e: last_log = f"خطأ غير متوقع أثناء التثبيت: {str(e)}" return redirect(url_for('index')) @app.route('/run/') def run_file(filename): global last_log filepath = os.path.join(UPLOAD_FOLDER, filename) try: # تشغيل السكربت في عملية منفصلة لتسجيل أي خطأ قد يحدث بسببه log_file_path = os.path.join(UPLOAD_FOLDER, f"{filename}.log") with open(log_file_path, 'w') as log_f: proc = subprocess.Popen(['python', filepath], stdout=log_f, stderr=subprocess.STDOUT, text=True) running_processes[filename] = proc last_log = f"🚀 تم إطلاق السكربت {filename} بنجاح في الخلفية!" except Exception as e: last_log = f"❌ خطأ في تشغيل السكربت {filename}: {str(e)}" return redirect(url_for('index')) @app.route('/stop/') def stop_file(filename): global last_log if filename in running_processes: try: running_processes[filename].terminate() running_processes[filename].wait(timeout=3) except: running_processes[filename].kill() del running_processes[filename] last_log = f"🛑 تم إيقاف السكربت: {filename}" return redirect(url_for('index')) @app.route('/delete/') def delete_file(filename): global last_log if filename in running_processes: try: running_processes[filename].terminate() except: pass del running_processes[filename] filepath = os.path.join(UPLOAD_FOLDER, filename) logpath = os.path.join(UPLOAD_FOLDER, f"{filename}.log") if os.path.exists(filepath): os.remove(filepath) if os.path.exists(logpath): os.remove(logpath) last_log = f"🗑️ تم حذف الملف: {filename} نهائياً." return redirect(url_for('index')) if __name__ == '__main__': print("السيرفر يعمل الآن على الرابط: http://127.0.0.1:5000") app.run(host='0.0.0.0', port=5000, debug=False)
