واجهة برمجة التطبيقات (API):

يمكنك استخدام الـ API لرفع الملفات برمجياً:

curl -X POST https://tmpweb.net/ \
  -F "website=@yourfile.zip" 
// مثال باستخدام JavaScript
async function uploadFile(file) {
    const formData = new FormData();
    formData.append('website', file);
    
    const response = await fetch('https://tmpweb.net/', {
        method: 'POST',
        body: formData
    });
    
    return await response.text();
}
